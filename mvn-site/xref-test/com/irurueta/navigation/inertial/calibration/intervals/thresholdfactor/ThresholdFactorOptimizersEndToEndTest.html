59      // Noise: accelerometer 100 micro-g/sqrt(Hz) and gyroscope 0.01 deg/sqrt(h) like in the paper (per sample
60      // standard deviation is rootPsd / sqrt(timeInterval)).
61      private static final double ACCEL_NOISE_ROOT_PSD = 100.0 * MICRO_G;
62      private static final double GYRO_NOISE_ROOT_PSD = 0.01 * DEG_TO_RAD / 60.0;

64      // Tolerances. Each static position is averaged over ~300 samples (per sample std ~7e-3 m/s^2), and 36-50
65      // positions are used. Maximum errors observed over the fixed seeds are ba 1.8e-4, Ma 3.9e-5, bg 3.6e-6 and
66      // Mg 4.4e-5, so tolerances are ~5 times larger than those.
67      private static final double MA_TOLERANCE = 2e-4;
68      private static final double BA_TOLERANCE = 1e-3;
69      private static final double MG_TOLERANCE = 2e-4;
70      private static final double BG_TOLERANCE = 2e-5;

107     @Test
108     void testExhaustiveOptimizerRecoversKnownParameters() throws Exception {
109         for (final var seed : SEEDS) {
110             generateSamples(seed);
111 
112             final var optimizer = new ExhaustiveAccelerometerAndGyroscopeIntervalDetectorThresholdFactorOptimizer(
113                     dataSource, createAccelerometerCalibrator(), createGyroscopeCalibrator());
114             final var thresholdFactor = optimizer.optimize();
115 
116             checkResult(optimizer, thresholdFactor);
117 
118             // optimization is deterministic for the same data
119             final var repeated = new ExhaustiveAccelerometerAndGyroscopeIntervalDetectorThresholdFactorOptimizer(
120                     dataSource, createAccelerometerCalibrator(), createGyroscopeCalibrator());
121             assertEquals(thresholdFactor, repeated.optimize(), 0.0);
122         }
123     }

138     @Test
139     void testFailingDataThrowsAndResetsResults() throws Exception {
140         generateSamples(SEEDS[0]);
141 
142         final var optimizer = new ExhaustiveAccelerometerAndGyroscopeIntervalDetectorThresholdFactorOptimizer(
143                 dataSource, createAccelerometerCalibrator(), createGyroscopeCalibrator());
144         optimizer.optimize();
145         assertTrue(optimizer.getThreshold() > 0.0);
146 
147         // too little data makes interval detection fail for every threshold factor (E2)
148         samples.clear();
149         assertThrows(IntervalDetectorThresholdFactorOptimizerException.class, optimizer::optimize);
150         assertEquals(0.0, optimizer.getOptimalThresholdFactor(), 0.0);
151         assertEquals(0.0, optimizer.getThreshold(), 0.0);
152 
153         generateSamples(SEEDS[0]);
154         final var bracketed = new BracketedAccelerometerAndGyroscopeIntervalDetectorThresholdFactorOptimizer(
155                 dataSource, createAccelerometerCalibrator(), createGyroscopeCalibrator());
156         bracketed.optimize();
157         samples.clear();
158         assertThrows(IntervalDetectorThresholdFactorOptimizerException.class, bracketed::optimize);
159         assertEquals(0.0, bracketed.getOptimalThresholdFactor(), 0.0);
160         assertEquals(0.0, bracketed.getThreshold(), 0.0);
161     }

163     @Test
164     void testNonFiniteGyroscopeMseIsNotAValidResult() throws Exception {
165         generateSamples(SEEDS[0]);
166 
167         // the default MSE rule only uses the accelerometer MSE, but a gyroscope calibration without a finite
168         // error must never be kept as a valid result
169         final var gyroscopeCalibrator = spy(createGyroscopeCalibrator());
170         doReturn(Double.NaN).when(gyroscopeCalibrator).getEstimatedMse();
171 
172         final var optimizer = new ExhaustiveAccelerometerAndGyroscopeIntervalDetectorThresholdFactorOptimizer(
173                 dataSource, createAccelerometerCalibrator(), gyroscopeCalibrator);
174         assertThrows(IntervalDetectorThresholdFactorOptimizerException.class, optimizer::optimize);
175         assertEquals(0.0, optimizer.getOptimalThresholdFactor(), 0.0);
176 
177         final var bracketed = new BracketedAccelerometerAndGyroscopeIntervalDetectorThresholdFactorOptimizer(
178                 dataSource, createAccelerometerCalibrator(), gyroscopeCalibrator);
179         assertThrows(IntervalDetectorThresholdFactorOptimizerException.class, bracketed::optimize);
180     }

182     @Test
183     void testInitialGyroscopeBiasIsUsedByDefault() throws Exception {
184         generateSamples(SEEDS[0]);
185 
186         final var gyroscopeCalibrator = createGyroscopeCalibrator();
187         final var optimizer = new ExhaustiveAccelerometerAndGyroscopeIntervalDetectorThresholdFactorOptimizer(
188                 dataSource, createAccelerometerCalibrator(), gyroscopeCalibrator);
189         assertTrue(optimizer.isInitialGyroscopeBiasUsed());
190         optimizer.optimize();
191 
192         // initial bias is the mean angular rate of the initial static period, which must be close to
193         // the true bias (E4): noise of the mean is ~3e-7 rad/s
194         final var bg = bg().getBuffer();
195         final var initialBias = gyroscopeCalibrator.getInitialBias();
196         assertNotNull(initialBias);
197         assertTrue(Math.abs(initialBias[0]) > 0.0);
198         assertArrayEquals(bg, initialBias, 5e-6);
199 
200         // when the option is disabled, the calibrator initial bias is kept
201         final var otherCalibrator = createGyroscopeCalibrator();
202         final var kept = new double[]{1e-3, -1e-3, 2e-3};
203         otherCalibrator.setInitialBias(kept);
204         final var other = new ExhaustiveAccelerometerAndGyroscopeIntervalDetectorThresholdFactorOptimizer(
205                 dataSource, createAccelerometerCalibrator(), otherCalibrator);
206         other.setInitialGyroscopeBiasUsed(false);
207         try {
208             other.optimize();
209         } catch (final IntervalDetectorThresholdFactorOptimizerException ignore) {
210             // calibration might fail with a wrong initial bias, which is irrelevant for this check
211         }
212         assertArrayEquals(kept, otherCalibrator.getInitialBias(), 0.0);
213     }

215     private void checkResult(
216             final AccelerometerAndGyroscopeIntervalDetectorThresholdFactorOptimizer optimizer,
217             final double thresholdFactor) throws WrongSizeException {
218         assertTrue(thresholdFactor >= 2.0 && thresholdFactor <= 10.0, "threshold factor " + thresholdFactor);
219         assertEquals(thresholdFactor, optimizer.getOptimalThresholdFactor(), 0.0);
220 
221         final var estimatedBa = Matrix.newFromArray(optimizer.getEstimatedAccelerometerBiases());
222         final var estimatedMa = optimizer.getEstimatedAccelerometerMa();
223         final var estimatedBg = Matrix.newFromArray(optimizer.getEstimatedGyroscopeBiases());
224         final var estimatedMg = optimizer.getEstimatedGyroscopeMg();
225 
226         assertTrue(ba().equals(estimatedBa, BA_TOLERANCE), "ba " + estimatedBa);
227         assertTrue(ma().equals(estimatedMa, MA_TOLERANCE), "ma " + estimatedMa);
228         assertTrue(bg().equals(estimatedBg, BG_TOLERANCE), "bg " + estimatedBg);
229         assertTrue(mg().equals(estimatedMg, MG_TOLERANCE), "mg " + estimatedMg);
230     }

276     /**
277      * Generates an initial static period (with gyroscope bias), followed by 36 to 50 static positions
278      * separated by rotations about random axes.
279      *
280      * @param seed seed of the random generator.
281      */
282     private void generateSamples(final long seed) throws WrongSizeException {
283         samples.clear();
284 
285         final var random = new Random(seed);
286         final var errors = new IMUErrors(ba(), bg(), ma(), mg(), new Matrix(3, 3), ACCEL_NOISE_ROOT_PSD,
287                 GYRO_NOISE_ROOT_PSD, 0.0, 0.0);
288 
289         // body to local rotation (row-major), initially level
290         var rotation = new double[]{1, 0, 0, 0, 1, 0, 0, 0, 1};
291         var index = 0;
292 
293         index = addStatic(INITIAL_STATIC_SAMPLES, rotation, errors, random, index);
294 
295         final var numPositions = 36 + random.nextInt(15);
296         for (var i = 0; i < numPositions; i++) {
297             index = addStatic(STATIC_SAMPLES, rotation, errors, random, index);
298             if (i < numPositions - 1) {
299                 // random unit axis (in body frame) and rotation angle between 60 and 170 degrees
300                 final var axis = new double[]{random.nextGaussian(), random.nextGaussian(), random.nextGaussian()};
301                 final var norm = Math.sqrt(axis[0] * axis[0] + axis[1] * axis[1] + axis[2] * axis[2]);
302                 for (var k = 0; k < 3; k++) {
303                     axis[k] /= norm;
304                 }
305                 final var angle = Math.toRadians(60.0 + 110.0 * random.nextDouble());
306                 rotation = addDynamic(rotation, axis, angle, errors, random, index);
307                 index += DYNAMIC_SAMPLES;
308             }
309         }
310     }
311 
312     private int addStatic(
313             final int numSamples, final double[] rotation, final IMUErrors errors, final Random random,
314             final int startIndex) {
315         var index = startIndex;
316         for (var i = 0; i < numSamples; i++, index++) {
317             add(index, trueKinematics(rotation, 0.0, 0.0, 0.0), errors, random);
318         }
319         return index;
320     }
321 
322     private double[] addDynamic(
323             final double[] startRotation, final double[] axis, final double angle, final IMUErrors errors,
324             final Random random, final int startIndex) {
325         var rotation = startRotation;
326         for (var i = 0; i < DYNAMIC_SAMPLES; i++) {
327             // smooth bump profile of angular rate, whose integral over the period equals the angle
328             final var p = (double) i / (double) DYNAMIC_SAMPLES;
329             final var s = Math.sin(Math.PI * p);
330             final var rate = 2.0 * angle / (DYNAMIC_SAMPLES * TIME_INTERVAL_SECONDS) * s * s;
331 
332             add(startIndex + i, trueKinematics(rotation, axis[0] * rate, axis[1] * rate, axis[2] * rate), errors,
333                     random);
334 
335             rotation = multiply(rotation, rodrigues(axis, rate * TIME_INTERVAL_SECONDS));
336         }
337         return rotation;
338     }
339 
340     private void add(final int index, final BodyKinematics truth, final IMUErrors errors, final Random random) {
341         final var measured = BodyKinematicsGenerator.generate(TIME_INTERVAL_SECONDS, truth, errors, random);
342         final var timed = new TimedBodyKinematics();
343         timed.setKinematics(measured);
344         timed.setTimestampSeconds(index * TIME_INTERVAL_SECONDS);
345         samples.add(timed);
346     }

348     /**
349      * Gets true kinematics: specific force is the reaction to gravity (local down) expressed in body frame, which
350      * is the third row of the body to local rotation matrix times -g.
351      */
352     private static BodyKinematics trueKinematics(
353             final double[] rotation, final double wx, final double wy, final double wz) {
354         final var result = new BodyKinematics();
355         result.setSpecificForceCoordinates(-GRAVITY * rotation[6], -GRAVITY * rotation[7], -GRAVITY * rotation[8]);
356         result.setAngularRateCoordinates(wx, wy, wz);
357         return result;
358     }

360     private static double[] rodrigues(final double[] axis, final double theta) {
361         final var c = Math.cos(theta);
362         final var s = Math.sin(theta);
363         final var t = 1.0 - c;
364         final var x = axis[0];
365         final var y = axis[1];
366         final var z = axis[2];
367         return new double[]{
368                 t * x * x + c, t * x * y - s * z, t * x * z + s * y,
369                 t * x * y + s * z, t * y * y + c, t * y * z - s * x,
370                 t * x * z - s * y, t * y * z + s * x, t * z * z + c
371         };
372     }
373 
374     private static double[] multiply(final double[] a, final double[] b) {
375         final var result = new double[9];
376         for (var i = 0; i < 3; i++) {
377             for (var j = 0; j < 3; j++) {
378                 var sum = 0.0;
379                 for (var k = 0; k < 3; k++) {
380                     sum += a[i * 3 + k] * b[k * 3 + j];
381                 }
382                 result[i * 3 + j] = sum;
383             }
384         }
385         return result;
386     }
