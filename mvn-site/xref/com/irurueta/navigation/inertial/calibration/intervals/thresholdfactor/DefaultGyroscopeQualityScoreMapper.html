16  package com.irurueta.navigation.inertial.calibration.intervals.thresholdfactor;
17  
18  import com.irurueta.navigation.inertial.calibration.BodyKinematicsSequence;
19  import com.irurueta.navigation.inertial.calibration.StandardDeviationTimedBodyKinematics;
20  
21  /**
22   * Default implementation to map a given measurement into a given quality score to be
23   * used for gyroscope calibration.
24   * <p>
25   * Gyroscope calibration integrates angular rates along each static-to-static motion sequence, so the
26   * integration error grows with the length of the sequence. Consequently, the quality score is computed as
27   * {@code 1 / (1 + duration)}, where duration is the time (in seconds) elapsed between the first and last
28   * items of the sequence. Longer sequences get strictly lower scores, and scores always belong to the
29   * range {@code (0, 1]}.
30   * <p>
31   * Previously, the score was computed from the average of the standard deviations of the items, which were
32   * the same for every sequence generated by a given sensor. That made the score practically constant and
33   * unable to discriminate between short and long sequences.
34   * <p>
35   * If the timestamps of the sequence are not usable (e.g. a single item or all items sharing the same
36   * timestamp), the sequence cannot be integrated and gets a zero score.
37   */
38  public class DefaultGyroscopeQualityScoreMapper implements
39          QualityScoreMapper<BodyKinematicsSequence<StandardDeviationTimedBodyKinematics>> {
40  
41      /**
42       * Maps a given value corresponding to a generated measurement, into a given quality score.
43       *
44       * @param value gyroscope sequence of standard deviation timed body kinematics items whose quality
45       *              score must be computed.
46       * @return mapped quality score, in the range (0, 1), or 0.0 if the sequence has no items or no positive
47       * duration.
48       */
49      @Override
50      public double map(final BodyKinematicsSequence<StandardDeviationTimedBodyKinematics> value) {
51          final var items = value.getSortedItems();
52          if (items == null || items.isEmpty()) {
53              return 0.0;
54          }
55  
56          var duration = items.getLast().getTimestampSeconds() - items.getFirst().getTimestampSeconds();
57          if (!Double.isFinite(duration) || duration <= 0.0) {
58              // a sequence without a positive duration cannot be integrated, so it gets the lowest score
59              return 0.0;
60          }
61  
62          return 1.0 / (1.0 + duration);
63      }
64  }
