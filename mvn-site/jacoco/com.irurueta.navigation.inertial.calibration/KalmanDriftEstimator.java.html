/**
 * Estimates accumulated drift in body orientation, position and velocity per
 * unit of time using an INS Kalman filtered solution.
 * The internal Kalman filter is fed with every provided (and optionally fixed)
 * body kinematics sample, and additionally with static position and velocity
 * fixes (position and velocity of the reference frame, the latter being zero for a
 * static body) once every epoch interval of
 * the Kalman filter, so that the filtered solution is corrected while the body
 * remains static.
 * Timestamps provided to the Kalman filter follow the convention
 * {@code (numberOfProcessedSamples + 1) * timeInterval}: the filter is initialized
 * at time zero and every sample is integrated over one time interval, so the
 * same number of samples is integrated as in a plain dead-reckoning
 * {@link DriftEstimator}.
 */
public class KalmanDriftEstimator extends DriftEstimator {

    /**
     * Configuration parameters for INS Loosely Coupled Kalman filter obtained
     * through calibration.
     */
    private INSLooselyCoupledKalmanConfig kalmanConfig;

    /**
     * Configuration parameters determining initial system noise covariance matrix.
     */
    private INSLooselyCoupledKalmanInitializerConfig initConfig;

    /**
     * Internal INS loosely coupled Kalman filtered estimator to estimate
     * accumulated body position, velocity and orientation.
     */
    private INSLooselyCoupledKalmanFilteredEstimator kalmanEstimator;

    /**
     * Current Kalman filter state.
     */
    private INSLooselyCoupledKalmanState state;

    /**
     * Reference position used as a static position fix of the Kalman filter.
     */
    private final ECEFPosition referenceEcefPosition = new ECEFPosition();

    /**
     * Reference velocity used as a static velocity fix of the Kalman filter.
     * This is the velocity of the reference frame (zero for a static body), which
     * is also the velocity that velocity drift is measured against.
     */
    private final ECEFVelocity referenceEcefVelocity = new ECEFVelocity();

    /**
     * Number of samples between consecutive static position and velocity fixes.
     */
    private int epochSamples = 1;

    /**
     * Relative tolerance used when converting the Kalman filter epoch interval
     * into a number of samples, so that an epoch interval that is an exact
     * multiple of the time interval is not rounded up because of floating-point
     * error.
     */
    private static final double EPOCH_SAMPLES_TOLERANCE = 1e-9;

    /**
     * Adds a sample of measured body kinematics (accelerometer and gyroscope readings)
     * obtained from an IMU, fixes their values and uses fixed values to estimate
     * current drift and their average values.
     * Every sample is integrated by the internal Kalman filter using timestamp
     * {@code (numberOfProcessedSamples + 1) * timeInterval}, and once every epoch
     * interval of the Kalman filter, a static position (reference frame position)
     * and velocity (reference frame velocity, zero for a static body) fix is provided to
     * it using the same timestamp.
     *
     * @param kinematics measured body kinematics.
     * @throws LockedException          if estimator is currently running.
     * @throws NotReadyException        if estimator is not ready.
     * @throws DriftEstimationException if estimation fails for some reason.
     */
    @Override
    public void addBodyKinematics(final BodyKinematics kinematics) throws LockedException, NotReadyException,
            DriftEstimationException {
        if (isRunning()) {
            throw new LockedException();
        }

        if (!isReady()) {
            throw new NotReadyException();
        }

        try {
            running = true;

            if (numberOfProcessedSamples == 0) {
                if (listener != null) {
                    listener.onStart(this);
                }

                initialize();
            }

            if (fixKinematics) {
                // fix kinematics and update kalman filter
                fixer.fix(kinematics, fixedKinematics);
            } else {
                // only update kalman filter
                fixedKinematics.copyFrom(kinematics);
            }

            final var sampleNumber = numberOfProcessedSamples + 1;
            final var timestamp = sampleNumber * timeInterval;
            kalmanEstimator.update(fixedKinematics, timestamp);

            if (sampleNumber % epochSamples == 0) {
                // static fix using position and velocity of reference frame. The number of
                // samples between fixes is never shorter than the epoch interval, so the
                // Kalman filter does not reject the epoch
                kalmanEstimator.updatePositionAndVelocity(referenceEcefPosition, referenceEcefVelocity, timestamp);
            }

            if (state == null) {
                state = kalmanEstimator.getState();
            } else {
                kalmanEstimator.getState(state);
            }

            // estimate drift values
            computeCurrentPositionDrift();
            computeCurrentVelocityDrift();
            computeCurrentOrientationDrift();

            numberOfProcessedSamples++;

            if (listener != null) {
                listener.onBodyKinematicsAdded(this, kinematics, fixedKinematics);
            }
        } catch (final AlgebraException | InvalidRotationMatrixException | INSException e) {
            throw new DriftEstimationException(e);
        } finally {
            running = false;
        }
    }

    /**
     * Resets this estimator to its initial state.
     *
     * @throws LockedException if estimator is currently running.
     */
    @Override
    public void reset() throws LockedException {
        if (isRunning()) {

        state = null;
        running = false;

        super.reset();
    }

    /**
     * Initializes internal frame and Kalman estimator when the first body kinematics
     * measurement is provided.
     * The Kalman filter is initialized at time zero so that the first sample is
     * integrated over one time interval, and the number of samples between static
     * position and velocity fixes is computed from the Kalman filter epoch interval
     * (rounded up to a whole number of samples, at least 1).
     *
     * @throws InvalidRotationMatrixException if orientation cannot be determined for
     *                                        numerical reasons.
     * @throws LockedException                if the Kalman estimator is locked.
     * @throws NotReadyException              if the Kalman estimator is not ready.
     * @throws INSException                   if the Kalman filter cannot be initialized.
     */
    private void initialize() throws InvalidRotationMatrixException, LockedException, NotReadyException,
            INSException {
        final var c = referenceFrame.getCoordinateTransformation();
        c.asRotation(refQ);
        refQ.inverse(invRefQ);

        frame.copyFrom(referenceFrame);

        referenceFrame.getECEFPosition(referenceEcefPosition);
        referenceFrame.getECEFVelocity(referenceEcefVelocity);

        kalmanEstimator = new INSLooselyCoupledKalmanFilteredEstimator(kalmanConfig, initConfig, frame);

        // initialize Kalman filter at time zero without kinematics
        kalmanEstimator.update(null, 0.0);

        final var epochInterval = kalmanEstimator.getEpochInterval();
        // round up (with a tolerance for exact multiples) so that fixes are never provided
        // before a whole epoch interval has elapsed, since the Kalman filter rejects them
        epochSamples = timeInterval > 0.0
                ? (int) Math.max(1.0, Math.min(Integer.MAX_VALUE,
                Math.ceil(epochInterval / timeInterval - EPOCH_SAMPLES_TOLERANCE)))
                : 1;
    }

    /**
     * Computes current position drift.
     */
    @Override
    protected void computeCurrentPositionDrift() {
        final var initX = referenceFrame.getX();
        final var initY = referenceFrame.getY();
        final var initZ = referenceFrame.getZ();

        final var currentX = state.getX();
        final var currentY = state.getY();
        final var currentZ = state.getZ();

        final var diffX = currentX - initX;
        final var diffY = currentY - initY;
        final var diffZ = currentZ - initZ;

        currentPositionDrift.setCoordinates(diffX, diffY, diffZ);

        currentPositionDriftMeters = currentPositionDrift.getNorm();
    }

    /**
     * Computes current velocity drift.
     */
    @Override
    protected void computeCurrentVelocityDrift() {
        final var initVx = referenceFrame.getVx();
        final var initVy = referenceFrame.getVy();
        final var initVz = referenceFrame.getVz();

        final var currentVx = state.getVx();
        final var currentVy = state.getVy();
        final var currentVz = state.getVz();

        final var diffVx = currentVx - initVx;
        final var diffVy = currentVy - initVy;
        final var diffVz = currentVz - initVz;

        currentVelocityDrift.setCoordinates(diffVx, diffVy, diffVz);

        currentVelocityDriftMetersPerSecond = currentVelocityDrift.getNorm();
    }

    /**
     * Computes current orientation drift.
     *
     * @throws InvalidRotationMatrixException if rotation cannot be accurately
     *                                        estimated.
     */
    @Override
    protected void computeCurrentOrientationDrift() throws InvalidRotationMatrixException {
        currentC = state.getBodyToEcefCoordinateTransformationMatrix();

        q.fromMatrix(currentC);
        q.combine(invRefQ);

        currentOrientationDriftRadians = q.getRotationAngle();
    }
